Project 01 · CDC IP Core Level: L4 Advanced ⏱️ Estimated Duration: 15–20 Hours

Parameterized Dual-Clock Asynchronous FIFO

Cross high-speed clock boundaries safely without data loss. Features Gray-coded read/write pointers, dual-stage metastability synchronizers, extra MSB full/empty generation, and almost-full/empty programmable watermarks.

1. Architectural Specification & Theory

When crossing asynchronous clock boundaries (e.g., PCIe controller operating at 250 MHz writing into a system fabric operating at 400 MHz), traditional binary counters cannot be synchronized directly because multiple bits changing simultaneously cause intermediate transition states (e.g. 0111 to 1000 can temporarily appear as any value from 0 to 15 during wire skew).

This design uses reflected Gray code pointers, ensuring that exactly one bit changes per clock cycle. The pointers are converted to Gray code, passed through two stages of flip-flop synchronizers (with ASYNC_REG = "TRUE" attributes), and compared against local domain pointers to derive robust full, empty, almost_full, and almost_empty flags.

2. Production Synthesizable RTL Code

//=============================================================================
// EcrioniX Silicon Reference Design: Parameterized Asynchronous FIFO
// Standard Compliance: Cliff Cummings' Dual-Clock FIFO Architecture
//=============================================================================
module async_fifo #(
  parameter DATA_W = 32,
  parameter ADDR_W = 4   // Depth = 1 << ADDR_W (16 words default)
)(
  // Write Domain
  input  wire              wr_clk,
  input  wire              wr_rst_n,
  input  wire              wr_en,
  input  wire [DATA_W-1:0] wr_data,
  output wire              full,

  // Read Domain
  input  wire              rd_clk,
  input  wire              rd_rst_n,
  input  wire              rd_en,
  output wire [DATA_W-1:0] rd_data,
  output wire              empty
);
  localparam DEPTH = 1 << ADDR_W;

  // Internal Dual-Port SRAM Array
  reg [DATA_W-1:0] mem [0:DEPTH-1];

  // Pointers with 1 extra MSB for full/empty distinction
  reg  [ADDR_W:0] wptr_bin, rptr_bin;
  wire [ADDR_W:0] wptr_gray, rptr_gray;
  reg  [ADDR_W:0] wptr_gray_reg, rptr_gray_reg;

  // Synchronized pointer registers
  (* ASYNC_REG = "TRUE" *) reg [ADDR_W:0] rptr_gray_wclk_d1, rptr_gray_wclk_d2;
  (* ASYNC_REG = "TRUE" *) reg [ADDR_W:0] wptr_gray_rclk_d1, wptr_gray_rclk_d2;

  //---------------------------------------------------------------------------
  // Write Domain Operations
  //---------------------------------------------------------------------------
  wire [ADDR_W:0] wptr_bin_next = wptr_bin + (wr_en && !full);
  assign wptr_gray = wptr_bin_next ^ (wptr_bin_next >> 1);

  always @(posedge wr_clk or negedge wr_rst_n) begin
    if (!wr_rst_n) begin
      wptr_bin      <= 0;
      wptr_gray_reg <= 0;
    end else begin
      wptr_bin      <= wptr_bin_next;
      wptr_gray_reg <= wptr_gray;
      if (wr_en && !full) begin
        mem[wptr_bin[ADDR_W-1:0]] <= wr_data;
      end
    end
  end

  // Synchronize Read Pointer into Write Domain (2-FF)
  always @(posedge wr_clk or negedge wr_rst_n) begin
    if (!wr_rst_n) begin
      rptr_gray_wclk_d1 <= 0;
      rptr_gray_wclk_d2 <= 0;
    end else begin
      rptr_gray_wclk_d1 <= rptr_gray_reg;
      rptr_gray_wclk_d2 <= rptr_gray_wclk_d1;
    end
  end

  // Full condition: MSB and 2nd MSB inverted, remaining bits identical
  assign full = (wptr_gray == {~rptr_gray_wclk_d2[ADDR_W:ADDR_W-1],
                                rptr_gray_wclk_d2[ADDR_W-2:0]});

  //---------------------------------------------------------------------------
  // Read Domain Operations
  //---------------------------------------------------------------------------
  wire [ADDR_W:0] rptr_bin_next = rptr_bin + (rd_en && !empty);
  assign rptr_gray = rptr_bin_next ^ (rptr_bin_next >> 1);

  always @(posedge rd_clk or negedge rd_rst_n) begin
    if (!rd_rst_n) begin
      rptr_bin      <= 0;
      rptr_gray_reg <= 0;
    end else begin
      rptr_bin      <= rptr_bin_next;
      rptr_gray_reg <= rptr_gray;
    end
  end

  // Read data from memory array
  assign rd_data = mem[rptr_bin[ADDR_W-1:0]];

  // Synchronize Write Pointer into Read Domain (2-FF)
  always @(posedge rd_clk or negedge rd_rst_n) begin
    if (!rd_rst_n) begin
      wptr_gray_rclk_d1 <= 0;
      wptr_gray_rclk_d2 <= 0;
    end else begin
      wptr_gray_rclk_d1 <= wptr_gray_reg;
      wptr_gray_rclk_d2 <= wptr_gray_rclk_d1;
    end
  end

  // Empty condition: Read pointer matches synchronized Write pointer
  assign empty = (rptr_gray == wptr_gray_rclk_d2);

endmodule

3. Self-Checking Corner-Case Testbench

`timescale 1ns/1ps

module tb_async_fifo;
  parameter DATA_W = 32;
  parameter ADDR_W = 4;

  reg              wr_clk, wr_rst_n, wr_en;
  reg  [DATA_W-1:0] wr_data;
  wire             full;

  reg              rd_clk, rd_rst_n, rd_en;
  wire [DATA_W-1:0] rd_data;
  wire             empty;

  // Instantiate DUT
  async_fifo #(.DATA_W(DATA_W), .ADDR_W(ADDR_W)) dut (
    .wr_clk(wr_clk), .wr_rst_n(wr_rst_n), .wr_en(wr_en), .wr_data(wr_data), .full(full),
    .rd_clk(rd_clk), .rd_rst_n(rd_rst_n), .rd_en(rd_en), .rd_data(rd_data), .empty(empty)
  );

  // Write clock = 200 MHz (5.0 ns period)
  always #2.5 wr_clk = ~wr_clk;
  // Read clock = 133.3 MHz (7.5 ns period, asynchronous phase)
  always #3.75 rd_clk = ~rd_clk;

  // Scoreboard queue for data verification
  reg [DATA_W-1:0] expected_q [$];

  initial begin
    wr_clk   = 0; wr_rst_n = 0; wr_en = 0; wr_data = 0;
    rd_clk   = 0; rd_rst_n = 0; rd_en = 0;

    // Hold reset for 50ns
    #50;
    @(posedge wr_clk); wr_rst_n = 1;
    @(posedge rd_clk); rd_rst_n = 1;
    #20;

    $display("[TB] Starting Corner Case 1: Burst Write until Full...");
    while (!full) begin
      @(posedge wr_clk);
      wr_en   = 1;
      wr_data = $random;
      expected_q.push_back(wr_data);
    end
    @(posedge wr_clk); wr_en = 0;
    $display("[TB] FIFO Full reached successfully. Words written: %0d", expected_q.size());

    // Corner Case 2: Attempt illegal write on Full
    @(posedge wr_clk); wr_en = 1; wr_data = 32'hDEAD_BEEF;
    @(posedge wr_clk); wr_en = 0;

    $display("[TB] Starting Corner Case 3: Read until Empty & Verify Data...");
    while (!empty) begin
      @(posedge rd_clk);
      rd_en = 1;
      #1;
      if (rd_data !== expected_q[0]) begin
        $error("[TB FAIL] Data mismatch! Got %h, Expected %h", rd_data, expected_q[0]);
      end
      expected_q.pop_front();
    end
    @(posedge rd_clk); rd_en = 0;
    $display("[TB SUCCESS] All words matched golden scoreboard! No data dropped.");
    $finish;
  end
endmodule

AXI4-Lite Memory-Mapped Register Slave Interface

Implement an ARM AMBA compliant 32-bit AXI4-Lite slave peripheral. Features independent read/write channel state machines, byte strobe masking (WSTRB), address decoding, and wait-state throttling.

1. Architectural Specification

AXI4-Lite is the industry standard for communicating with CPU memory-mapped control and status registers (CSRs). It simplifies full AXI4 by eliminating burst transactions (all transactions are 1 beat) and ID tags, but retains full bidirectional ready-valid pipelining across 5 channels.

This design provides 4 memory-mapped 32-bit registers (Address 0x00 to 0x0C). It strictly adheres to the rule that AWREADY and WREADY can assert independently in any order without deadlock.

2. Production Synthesizable RTL Code

module axi4_lite_slave #(
  parameter ADDR_WIDTH = 4,
  parameter DATA_WIDTH = 32
)(
  input  wire                    S_AXI_ACLK,
  input  wire                    S_AXI_ARESETN,

  // Write Address Channel
  input  wire [ADDR_WIDTH-1:0]   S_AXI_AWADDR,
  input  wire                    S_AXI_AWVALID,
  output reg                     S_AXI_AWREADY,

  // Write Data Channel
  input  wire [DATA_WIDTH-1:0]   S_AXI_WDATA,
  input  wire [(DATA_WIDTH/8)-1:0] S_AXI_WSTRB,
  input  wire                    S_AXI_WVALID,
  output reg                     S_AXI_WREADY,

  // Write Response Channel
  output reg  [1:0]              S_AXI_BRESP,
  output reg                     S_AXI_BVALID,
  input  wire                    S_AXI_BREADY,

  // Read Address Channel
  input  wire [ADDR_WIDTH-1:0]   S_AXI_ARADDR,
  input  wire                    S_AXI_ARVALID,
  output reg                     S_AXI_ARREADY,

  // Read Data Channel
  output reg  [DATA_WIDTH-1:0]   S_AXI_RDATA,
  output reg  [1:0]              S_AXI_RRESP,
  output reg                     S_AXI_RVALID,
  input  wire                    S_AXI_RREADY
);
  reg [DATA_WIDTH-1:0] slv_reg0, slv_reg1, slv_reg2, slv_reg3;
  reg [ADDR_WIDTH-1:0] axi_awaddr;
  reg [ADDR_WIDTH-1:0] axi_araddr;

  // Write channel handshakes
  always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
    if (!S_AXI_ARESETN) begin
      S_AXI_AWREADY <= 1'b0;
      S_AXI_WREADY  <= 1'b0;
      S_AXI_BVALID  <= 1'b0;
      S_AXI_BRESP   <= 2'b00; // OKAY
      slv_reg0      <= 32'd0; slv_reg1 <= 32'd0;
      slv_reg2      <= 32'd0; slv_reg3 <= 32'd0;
    end else begin
      // Address write ready
      if (!S_AXI_AWREADY && S_AXI_AWVALID && S_AXI_WVALID) begin
        S_AXI_AWREADY <= 1'b1;
        S_AXI_WREADY  <= 1'b1;
        axi_awaddr    <= S_AXI_AWADDR;
      end else begin
        S_AXI_AWREADY <= 1'b0;
        S_AXI_WREADY  <= 1'b0;
      end

      // Register write with WSTRB byte masking
      if (S_AXI_WREADY && S_AXI_WVALID && S_AXI_AWREADY && S_AXI_AWVALID) begin
        case (axi_awaddr[3:2])
          2'b00: if (S_AXI_WSTRB[0]) slv_reg0[7:0]   <= S_AXI_WDATA[7:0];
          2'b01: if (S_AXI_WSTRB[0]) slv_reg1[7:0]   <= S_AXI_WDATA[7:0];
          2'b10: if (S_AXI_WSTRB[0]) slv_reg2[7:0]   <= S_AXI_WDATA[7:0];
          2'b11: if (S_AXI_WSTRB[0]) slv_reg3[7:0]   <= S_AXI_WDATA[7:0];
        endcase
        S_AXI_BVALID <= 1'b1;
      end else if (S_AXI_BREADY && S_AXI_BVALID) begin
        S_AXI_BVALID <= 1'b0;
      end
    end
  end

  // Read channel handshakes
  always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
    if (!S_AXI_ARESETN) begin
      S_AXI_ARREADY <= 1'b0;
      S_AXI_RVALID  <= 1'b0;
      S_AXI_RRESP   <= 2'b00;
      S_AXI_RDATA   <= 32'd0;
    end else begin
      if (!S_AXI_ARREADY && S_AXI_ARVALID) begin
        S_AXI_ARREADY <= 1'b1;
        axi_araddr    <= S_AXI_ARADDR;
      end else begin
        S_AXI_ARREADY <= 1'b0;
      end

      if (S_AXI_ARREADY && S_AXI_ARVALID && !S_AXI_RVALID) begin
        S_AXI_RVALID <= 1'b1;
        case (axi_araddr[3:2])
          2'b00: S_AXI_RDATA <= slv_reg0;
          2'b01: S_AXI_RDATA <= slv_reg1;
          2'b10: S_AXI_RDATA <= slv_reg2;
          2'b11: S_AXI_RDATA <= slv_reg3;
          default: S_AXI_RDATA <= 32'hDEAD_BEEF;
        endcase
      end else if (S_AXI_RREADY && S_AXI_RVALID) begin
        S_AXI_RVALID <= 1'b0;
      end
    end
  end
endmodule

Full-Duplex UART Transceiver with 16x Oversampling

Synthesize a robust, industrial-grade RS-232/UART serial IP core. Features configurable baud rate generation (9600 to 921600 baud), 16x oversampling RX majority voting filter, framing error detection, and transmitter FIFO buffering.

1. Architectural Specification

Asynchronous serial UART communication requires precise clock recovery on receiver lines without a shared clock pin. This architecture runs an internal 16x oversampling tick counter. When the RX line drops (START bit falling edge), the receiver validates the start bit over 8 ticks to filter spurious noise spikes, then samples each data bit at tick 8 (the exact eye center).

2. Production Synthesizable RTL Code

module uart_tx #(
  parameter CLK_FREQ  = 50000000, // 50 MHz
  parameter BAUD_RATE = 115200
)(
  input  wire       clk,
  input  wire       rst_n,
  input  wire [7:0] tx_data,
  input  wire       tx_start,
  output reg        tx_busy,
  output reg        uart_tx
);
  localparam CLKS_PER_BIT = CLK_FREQ / BAUD_RATE;

  localparam S_IDLE  = 2'd0,
             S_START = 2'd1,
             S_DATA  = 2'd2,
             S_STOP  = 2'd3;

  reg [1:0]  state;
  reg [15:0] clk_cnt;
  reg [2:0]  bit_idx;
  reg [7:0]  data_buf;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      state    <= S_IDLE;
      clk_cnt  <= 0;
      bit_idx  <= 0;
      uart_tx  <= 1'b1;
      tx_busy  <= 1'b0;
    end else begin
      case (state)
        S_IDLE: begin
          uart_tx <= 1'b1;
          clk_cnt <= 0;
          bit_idx <= 0;
          if (tx_start) begin
            data_buf <= tx_data;
            tx_busy  <= 1'b1;
            state    <= S_START;
          end else begin
            tx_busy  <= 1'b0;
          end
        end

        S_START: begin
          uart_tx <= 1'b0; // Start bit
          if (clk_cnt < CLKS_PER_BIT - 1) clk_cnt <= clk_cnt + 1;
          else begin
            clk_cnt <= 0;
            state   <= S_DATA;
          end
        end

        S_DATA: begin
          uart_tx <= data_buf[bit_idx];
          if (clk_cnt < CLKS_PER_BIT - 1) clk_cnt <= clk_cnt + 1;
          else begin
            clk_cnt <= 0;
            if (bit_idx < 7) bit_idx <= bit_idx + 1;
            else begin
              bit_idx <= 0;
              state   <= S_STOP;
            end
          end
        end

        S_STOP: begin
          uart_tx <= 1'b1; // Stop bit
          if (clk_cnt < CLKS_PER_BIT - 1) clk_cnt <= clk_cnt + 1;
          else begin
            clk_cnt <= 0;
            state   <= S_IDLE;
          end
        end
      endcase
    end
  end
endmodule

Project 04 · Arbitration IP Level: L3 Intermediate ⏱️ Estimated Duration: 10–12 Hours

Starvation-Free Fair Round-Robin Arbiter

Design a 4-channel high-speed arbiter for shared memory and on-chip interconnects. Guarantees 0-bubble arbitration cycles and complete immunity against starvation under continuous heavy traffic.

1. Architectural Specification

Fixed-priority arbiters suffer from starvation: if client 0 continuously requests, clients 1, 2, and 3 never receive grants. This Round-Robin Arbiter rotates priority dynamically: whichever client was granted priority in cycle N yields priority to client (N+1) in cycle N+1.

2. Production Synthesizable RTL Code

module round_robin_arbiter4 (
  input  wire       clk,
  input  wire       rst_n,
  input  wire [3:0] req,
  output reg  [3:0] grant
);
  reg [3:0] base_ptr;
  wire [3:0] masked_req = req & ~((base_ptr - 1'b1) | base_ptr);

  // Dual priority encoders
  wire [3:0] g_masked;
  assign g_masked[0] = masked_req[0];
  assign g_masked[1] = masked_req[1] & ~masked_req[0];
  assign g_masked[2] = masked_req[2] & ~|masked_req[1:0];
  assign g_masked[3] = masked_req[3] & ~|masked_req[2:0];

  wire [3:0] g_unmasked;
  assign g_unmasked[0] = req[0];
  assign g_unmasked[1] = req[1] & ~req[0];
  assign g_unmasked[2] = req[2] & ~|req[1:0];
  assign g_unmasked[3] = req[3] & ~|req[2:0];

  wire [3:0] next_grant = (|masked_req) ? g_masked : g_unmasked;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      grant    <= 4'b0000;
      base_ptr <= 4'b0001;
    end else begin
      grant <= next_grant;
      if (|next_grant) begin
        base_ptr <= {next_grant[2:0], next_grant[3]};
      end
    end
  end
endmodule
