module top_module (
  input  wire        clk,
  input  wire        rst_n,
  input  wire        s_valid,
  output wire        s_ready,
  input  wire [63:0] s_data,
  input  wire        s_last,
  output wire        m_valid,
  input  wire        m_ready,
  output wire [63:0] m_data,
  output wire        m_last,
  output reg  [2:0]  buffer_occupancy
);

  reg [64:0] mem [0:3];
  reg [1:0] wr_ptr, rd_ptr;

  assign s_ready = (buffer_occupancy < 3'd4);
  assign m_valid = (buffer_occupancy > 3'd0);

  wire [64:0] rd_entry = mem[rd_ptr];
  assign m_data = rd_entry[63:0];
  assign m_last = rd_entry[64];

  wire push = s_valid && s_ready;
  wire pop  = m_valid && m_ready;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      wr_ptr           <= 2'd0;
      rd_ptr           <= 2'd0;
      buffer_occupancy <= 3'd0;
    end else begin
      if (push) begin
        mem[wr_ptr] <= {s_last, s_data};
        wr_ptr      <= wr_ptr + 2'd1;
      end

      if (pop) begin
        rd_ptr <= rd_ptr + 2'd1;
      end

      case ({push, pop})
        2'b10: buffer_occupancy <= buffer_occupancy + 3'd1;
        2'b01: buffer_occupancy <= buffer_occupancy - 3'd1;
        default: ; // 2'b00 or 2'b11 -> unchanged
      endcase
    end
  end

endmodule

Architectural Deep Dive & Implementation Walkthrough
Elastic Buffering in Multi-Die Systems


In multi-chiplet packaging, on-die protocols (AXI-Stream, AMBA CHI) operate at different bus widths and clock rates than off-die D2D physical layers. Streaming Protocol Bridges isolate these interfaces using elastic FIFO buffers.


Key Operational Guarantees


    
  
  • Cut-Through Latency: Data is available for reading as soon as it is buffered.
    • 
  
  • Backpressure Absorption: If downstream temporarily deasserts m_ready, the bridge absorbs up to 4 flits before raising backpressure upstream (s_ready = 0).
    • 
  
  • Framing Preservation: The end-of-packet boundary (TLAST / flit trailer) remains aligned with the payload.
    •

⚠️ Common Mistakes to Avoid
Failing to allow simultaneous push and pop when buffer_occupancy is 4, which artificially reduces throughput.
