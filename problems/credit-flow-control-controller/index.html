module top_module (
  input  wire        clk,
  input  wire        rst_n,
  input  wire        tx_valid,
  input  wire [31:0] tx_data,
  output wire        tx_ready,
  output reg         phy_tx_valid,
  output reg  [31:0] phy_tx_data,
  input  wire        credit_return_in,
  output reg  [3:0]  credits_available
);

  assign tx_ready = (credits_available > 4'd0);
  wire send_beat = tx_valid && tx_ready;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      credits_available <= 4'd8;
      phy_tx_valid      <= 1'b0;
      phy_tx_data       <= 32'h0;
    end else begin
      phy_tx_valid <= send_beat;
      if (send_beat) begin
        phy_tx_data <= tx_data;
      end

      case ({send_beat, credit_return_in})
        2'b10: begin // Sent, no return -> decrement
          if (credits_available > 4'd0)
            credits_available <= credits_available - 4'd1;
        end
        2'b01: begin // No send, return -> increment
          if (credits_available < 4'd15)
            credits_available <= credits_available + 4'd1;
        end
        2'b11: begin // Simultaneous send and return -> net zero change
          credits_available <= credits_available;
        end
        default: ; // 2'b00 -> hold
      endcase
    end
  end

endmodule

Architectural Deep Dive & Implementation Walkthrough
Why Chiplet Links Use Credit Flow Control


In on-chip buses (like AXI), valid-ready backpressure requires the consumer to immediately assert READY = 0 to halt the producer. However, across a multi-die package (such as UCIe or optical chiplet interconnects), signal propagation latency can span multiple clock cycles.


If a receiver ran out of buffer space, its READY = 0 backpressure would arrive too late, causing buffer overflow and packet loss. Credit-based flow control solves this by granting the transmitter a known allowance of credits corresponding to physical buffer slots. The transmitter only sends when it possesses a credit, guaranteeing zero buffer overflows regardless of physical distance.


      
Common RTL Designer Pitfalls
⚠️ Common Mistakes to Avoid
Failing to handle simultaneous transmit and credit return, which causes credit loss or artificial stalls.


      
Silicon & Hardware Engineering Notes
🛠️ ASIC vs FPGA Hardware Considerations
Universal Chiplet Interconnect Express (UCIe) Flit Transfer protocols rely on credit-based flow control between the D2D Adapter and the remote die Die-to-Die controller.
