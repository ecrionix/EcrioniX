Design a 4-channel AXI-Stream arbiter featuring packet-boundary locking, starvation-free fair round-robin scheduling, and full backpressure propagation.

module top_module (
  input  wire         aclk,
  input  wire         aresetn,
  input  wire [3:0]   s_tvalid,
  output reg  [3:0]   s_tready,
  input  wire [127:0] s_tdata,
  input  wire [3:0]   s_tlast,
  output reg          m_tvalid,
  input  wire         m_tready,
  output reg  [31:0]  m_tdata,
  output reg          m_tlast
);

  reg [1:0] winner;
  reg [1:0] base_ptr;
  reg       locked;

  wire [31:0] ch_data [0:3];
  assign ch_data[0] = s_tdata[31:0];
  assign ch_data[1] = s_tdata[63:32];
  assign ch_data[2] = s_tdata[95:64];
  assign ch_data[3] = s_tdata[127:96];

  // Combinational round-robin priority picker
  reg [1:0] next_grant;
  reg       any_req;

  always @(*) begin
    any_req    = |s_tvalid;
    next_grant = base_ptr;
    if (s_tvalid[base_ptr]) begin
      next_grant = base_ptr;
    end else if (s_tvalid[(base_ptr + 2'd1) & 2'd3]) begin
      next_grant = (base_ptr + 2'd1) & 2'd3;
    end else if (s_tvalid[(base_ptr + 2'd2) & 2'd3]) begin
      next_grant = (base_ptr + 2'd2) & 2'd3;
    end else if (s_tvalid[(base_ptr + 2'd3) & 2'd3]) begin
      next_grant = (base_ptr + 2'd3) & 2'd3;
    end
  end

  // Mux selection
  always @(*) begin
    if (locked || any_req) begin
      m_tvalid = s_tvalid[winner];
      m_tdata  = ch_data[winner];
      m_tlast  = s_tlast[winner];
    end else begin
      m_tvalid = 1'b0;
      m_tdata  = 32'h0;
      m_tlast  = 1'b0;
    end

    s_tready = 4'b0000;
    if (locked || any_req) begin
      s_tready[winner] = m_tready;
    end
  end

  always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
      winner   <= 2'd0;
      base_ptr <= 2'd0;
      locked   <= 1'b0;
    end else begin
      if (!locked) begin
        if (any_req) begin
          winner <= next_grant;
          locked <= 1'b1;
          // Single-beat packet check
          if (s_tvalid[next_grant] && m_tready && s_tlast[next_grant]) begin
            locked   <= 1'b0;
            base_ptr <= (next_grant + 2'd1) & 2'd3;
          end
        end
      end else begin
        // Locked in transfer
        if (m_tvalid && m_tready && m_tlast) begin
          locked   <= 1'b0;
          base_ptr <= (winner + 2'd1) & 2'd3;
        end
      end
    end
  end

endmodule

Architectural Deep Dive & Implementation Walkthrough
Packet Interleaving vs. Packet Locking


In streaming networking protocols (Ethernet, PCIe TLP, AXI-Stream), data beats belong to distinct packets bounded by TLAST. Arbitrating on a pure per-beat basis would cause packet fragmentation and interleaving, corrupting packet payloads at the receiver.


Arbiter Design Rules


    
  
  • Arbitration Phase: When the arbiter is idle (unlocked), it inspects all requesting channels and grants access according to the current round-robin priority pointer.
    • 
  
  • Locked Phase: The winner gains exclusive access to the egress bus. Even if downstream backpressure occurs (m_tready = 0), the arbiter holds its grant.
    • 
  
  • Release & Rotation: When the final beat is transferred (m_tvalid && m_tready && m_tlast), the lock releases and the priority pointer advances to prevent starvation.
    • 



      
Common RTL Designer Pitfalls
⚠️ Common Mistakes to Avoid
Switching channels on backpressure (m_tready=0) or breaking packet boundaries when higher-priority requesters arrive.


      
Silicon & Hardware Engineering Notes
🛠️ ASIC vs FPGA Hardware Considerations
In network-on-chip (NoC) routers, packet arbiters include aging counters to prevent low-bandwidth channels from starving under high-rate streaming flows.
