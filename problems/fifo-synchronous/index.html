module top_module(
  input        clk,
  input        areset_n,
  input        wr_en,
  input  [7:0] din,
  input        rd_en,
  output reg [7:0] dout,
  output       full,
  output       empty
);
  reg [7:0] mem [0:7];
  reg [2:0] wr_ptr;
  reg [2:0] rd_ptr;
  reg [3:0] count;

  assign full  = (count == 4'd8);
  assign empty = (count == 4'd0);

  wire write_valid = wr_en && !full;
  wire read_valid  = rd_en && !empty;

  always @(posedge clk or negedge areset_n) begin
    if (!areset_n) begin
      wr_ptr <= 3'd0;
      rd_ptr <= 3'd0;
      count  <= 4'd0;
      dout   <= 8'd0;
    end else begin
      if (write_valid) begin
        mem[wr_ptr] <= din;
        wr_ptr      <= wr_ptr + 1'b1;
      end
      if (read_valid) begin
        dout   <= mem[rd_ptr];
        rd_ptr <= rd_ptr + 1'b1;
      end

      case ({write_valid, read_valid})
        2'b10: count <= count + 1'b1;
        2'b01: count <= count - 1'b1;
        default: count <= count; // 00 or 11 (simultaneous read/write maintains occupancy)
      endcase
    end
  end
endmodule

Architectural Deep Dive & Implementation Walkthrough
A synchronous FIFO acts as an elastic pipeline buffer between modules on the same clock. Write and read pointers wrap around circularly through memory addresses $0 \dots 7$. An occupancy counter tracks buffer fullness, preventing write overflows when full and read underflows when empty.

⚠️ Common Mistakes to Avoid
