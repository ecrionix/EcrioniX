Verified Synthesizable RTL 
module top_module #(
  parameter DATA_WIDTH = 8
)(
  input                    clk,
  input                    areset_n,
  input                    s_valid,
  output reg               s_ready,
  input   [DATA_WIDTH-1:0] s_data,
  output reg               m_valid,
  input                    m_ready,
  output reg [DATA_WIDTH-1:0] m_data
);
  reg [DATA_WIDTH-1:0] skid_data;
  reg                  skid_valid;

  always @(posedge clk or negedge areset_n) begin
    if (!areset_n) begin
      s_ready    <= 1'b1;
      m_valid    <= 1'b0;
      m_data     <= {DATA_WIDTH{1'b0}};
      skid_valid <= 1'b0;
      skid_data  <= {DATA_WIDTH{1'b0}};
    end else begin
      if (m_ready) begin
        if (skid_valid) begin
          m_valid    <= 1'b1;
          m_data     <= skid_data;
          skid_valid <= 1'b0;
          s_ready    <= 1'b1;
        end else if (s_valid && s_ready) begin
          m_valid    <= 1'b1;
          m_data     <= s_data;
          s_ready    <= 1'b1;
        end else begin
          m_valid    <= 1'b0;
          s_ready    <= 1'b1;
        end
      end else begin // m_ready is low (downstream stalled)
        if (s_valid && s_ready) begin
          skid_data  <= s_data;
          skid_valid <= 1'b1;
          s_ready    <= 1'b0; // can no longer accept when skid is full
        end
      end
    end
  end
endmodule

Architectural Deep Dive & Implementation Walkthrough
In long AXI streaming interconnects, backpressure paths (`ready`) create combinational paths spanning across entire chips. A standard register slice cuts forward paths but leaves `ready` combinational. A Skid Buffer cuts BOTH forward and backward timing paths, isolating timing domains while sustaining 100% throughput with 0 bubble cycles.
