module top_module (
  input  wire        aclk,
  input  wire        aresetn,
  input  wire        s_tvalid,
  output reg         s_tready,
  input  wire [31:0] s_tdata,
  input  wire [3:0]  s_tkeep,
  input  wire        s_tlast,
  output reg         m_tvalid,
  input  wire        m_tready,
  output reg  [31:0] m_tdata,
  output reg  [3:0]  m_tkeep,
  output reg         m_tlast
);

  // Skid register storage
  reg [31:0] skid_data;
  reg [3:0]  skid_keep;
  reg        skid_last;
  reg        skid_valid;

  always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
      s_tready   <= 1'b1;
      m_tvalid   <= 1'b0;
      m_tdata    <= 32'h0;
      m_tkeep    <= 4'h0;
      m_tlast    <= 1'b0;
      skid_valid <= 1'b0;
      skid_data  <= 32'h0;
      skid_keep  <= 4'h0;
      skid_last  <= 1'b0;
    end else begin
      if (m_tvalid && !m_tready) begin
        // Downstream stalled: capture into skid buffer if new valid beat arrives
        if (s_tvalid && s_tready) begin
          skid_data  <= s_tdata;
          skid_keep  <= s_tkeep;
          skid_last  <= s_tlast;
          skid_valid <= 1'b1;
          s_tready   <= 1'b0; // Skid full: assert backpressure upstream
        end
      end else begin
        // Downstream ready: drain skid if present, otherwise pass from upstream
        if (skid_valid) begin
          m_tvalid   <= 1'b1;
          m_tdata    <= skid_data;
          m_tkeep    <= skid_keep;
          m_tlast    <= skid_last;
          skid_valid <= 1'b0;
          s_tready   <= 1'b1;
        end else if (s_tvalid && s_tready) begin
          m_tvalid <= 1'b1;
          m_tdata  <= s_tdata;
          m_tkeep  <= s_tkeep;
          m_tlast  <= s_tlast;
          s_tready <= 1'b1;
        end else begin
          m_tvalid <= 1'b0;
          s_tready <= 1'b1;
        end
      end
    end
  end

endmodule

Architectural Deep Dive & Implementation Walkthrough
Why Chips Need Skid Buffers


In high-speed physical design (1 GHz+), long routing wires between distant IP blocks cannot meet setup time if combinational control signals (like READY backpressure) traverse multiple hierarchical boundaries.


The Register Slice Dilemma


If you register forward signals (VALID/DATA) but pass READY combinationally, the reverse timing path remains broken. If you naively register READY, the upstream transmitter cannot know the downstream stalled until 1 cycle later, dropping a packet or halving bandwidth to 50%.


The Skid Buffer introduces a secondary holding register that catches the extra in-flight beat during the 1-cycle round-trip latency of the registered s_tready signal, enabling 100% throughput and complete timing closure.


      
Common RTL Designer Pitfalls
⚠️ Common Mistakes to Avoid
Failing to transfer the skid buffer payload to m_tdata immediately when m_tready goes high, leading to out-of-order data or pipeline stalls.


      
Silicon & Hardware Engineering Notes
🛠️ ASIC vs FPGA Hardware Considerations
Standard industry AXI-Stream infrastructure IPs (such as Xilinx AXI4-Stream Register Slice or Arm CoreLink) use skid buffers at almost every major clock domain or floorplan SLR boundary.
