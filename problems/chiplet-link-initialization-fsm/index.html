module top_module (
  input  wire        clk,
  input  wire        rst_n,
  input  wire        init_start,
  input  wire        partner_req,
  input  wire        partner_ack,
  input  wire        train_pattern_ok,
  input  wire        error_detected,
  output reg         link_req_out,
  output reg         link_ack_out,
  output wire        link_active,
  output wire        link_error,
  output reg  [2:0]  state_out
);

  localparam ST_RESET    = 3'd0;
  localparam ST_REQ      = 3'd1;
  localparam ST_TRAIN    = 3'd2;
  localparam ST_CONFIG   = 3'd3;
  localparam ST_ACTIVE   = 3'd4;
  localparam ST_ERROR    = 3'd5;
  localparam ST_RECOVERY = 3'd6;

  reg [3:0] timeout_cnt;

  assign link_active = (state_out == ST_ACTIVE);
  assign link_error  = (state_out == ST_ERROR);

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      state_out    <= ST_RESET;
      link_req_out <= 1'b0;
      link_ack_out <= 1'b0;
      timeout_cnt  <= 4'd0;
    end else begin
      case (state_out)
        ST_RESET: begin
          link_req_out <= 1'b0;
          link_ack_out <= 1'b0;
          timeout_cnt  <= 4'd0;
          if (init_start) begin
            state_out    <= ST_REQ;
            link_req_out <= 1'b1;
          end
        end

        ST_REQ: begin
          link_req_out <= 1'b1;
          if (partner_ack) begin
            state_out    <= ST_TRAIN;
            link_ack_out <= 1'b1;
            timeout_cnt  <= 4'd0;
          end
        end

        ST_TRAIN: begin
          if (train_pattern_ok) begin
            state_out   <= ST_CONFIG;
            timeout_cnt <= 4'd0;
          end else if (timeout_cnt >= 4'd7) begin
            state_out   <= ST_ERROR;
            timeout_cnt <= 4'd0;
          end else begin
            timeout_cnt <= timeout_cnt + 4'd1;
          end
        end

        ST_CONFIG: begin
          link_ack_out <= 1'b1;
          if (partner_req) begin
            state_out <= ST_ACTIVE;
          end
        end

        ST_ACTIVE: begin
          link_req_out <= 1'b1;
          link_ack_out <= 1'b1;
          if (error_detected) begin
            state_out <= ST_ERROR;
          end
        end

        ST_ERROR: begin
          link_req_out <= 1'b0;
          link_ack_out <= 1'b0;
          state_out    <= ST_RECOVERY;
        end

        ST_RECOVERY: begin
          state_out    <= ST_REQ;
          link_req_out <= 1'b1;
        end

        default: state_out <= ST_RESET;
      endcase
    end
  end

endmodule

Architectural Deep Dive & Implementation Walkthrough
UCIe and Multi-Die Link Bring-Up


Modern advanced packaging (TSMC CoWoS, Intel EMIB) connects multiple silicon dies through micro-bumps. Before high-speed main data lanes (32 Gbps+) can transmit, a low-speed Sideband Channel brings up and trains the interface.


Educational State Machine Overview


    
  
  • Phase 1: Discovery (REQ / ACK): Dies signal presence across dedicated sideband wires.
    • 
  
  • Phase 2: PHY Training: Pseudorandom bit sequences (PRBS) calibrate receiver equalizers and per-lane delay locks.
    • 
  
  • Phase 3: Parameter Configuration: Negotiate lane count, clock frequency, and CRC algorithms.
    • 
  
  • Phase 4: Active Operation & Recovery: Upon parity fault or signal loss, the controller enters RECOVERY to re-lock PHY equalizers without resetting the host CPU.
    • 



Note: This module implements an educational abstraction of multi-die link training principles; actual UCIe specifications define detailed sub-states for lane repair and power management.


      
Common RTL Designer Pitfalls
⚠️ Common Mistakes to Avoid
Staying in ST_TRAIN forever without a timeout mechanism, which locks up the chip if partner die fails to power up.


      
Silicon & Hardware Engineering Notes
🛠️ ASIC vs FPGA Hardware Considerations
Universal Chiplet Interconnect Express (UCIe) incorporates redundant spare micro-bumps per cluster that are dynamically swapped in during the training phase if physical faults are detected.
