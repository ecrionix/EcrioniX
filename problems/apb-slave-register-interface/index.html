module top_module(
  input         pclk,
  input         preset_n,
  input  [3:0]  paddr,
  input         psel,
  input         penable,
  input         pwrite,
  input  [31:0] pwdata,
  output reg [31:0] prdata,
  output        pready
);
  reg [31:0] reg0;
  reg [31:0] reg1;

  assign pready = 1'b1; // zero wait-state response

  // Write phase
  always @(posedge pclk or negedge preset_n) begin
    if (!preset_n) begin
      reg0 <= 32'h0;
      reg1 <= 32'h0;
    end else if (psel && penable && pwrite) begin
      case (paddr[3:0])
        4'h0: reg0 <= pwdata;
        4'h4: reg1 <= pwdata;
      endcase
    end
  end

  // Read phase
  always @(*) begin
    if (psel && !pwrite) begin
      case (paddr[3:0])
        4'h0: prdata = reg0;
        4'h4: prdata = reg1;
        default: prdata = 32'h0;
      endcase
    end else begin
      prdata = 32'h0;
    end
  end
endmodule

Architectural Deep Dive & Implementation Walkthrough
AMBA APB is an unpipelined, low-power control bus designed by ARM for peripheral register access (UART, Timers, GPIO). Each transaction consists of two cycles: 1) SETUP cycle (`psel=1, penable=0`), where address and write data are driven; 2) ACCESS cycle (`penable=1`), where data is sampled or driven.

⚠️ Common Mistakes to Avoid
