module top_module(
  input        clk,
  input        areset_n,
  input        tx_start,
  input  [7:0] tx_data,
  output reg   tx,
  output       busy
);
  reg [3:0] bit_cnt;
  reg [7:0] sh_reg;

  assign busy = (bit_cnt != 4'd0);

  always @(posedge clk or negedge areset_n) begin
    if (!areset_n) begin
      tx      <= 1'b1; // Idle line is HIGH
      bit_cnt <= 4'd0;
      sh_reg  <= 8'd0;
    end else if (bit_cnt == 4'd0) begin
      tx <= 1'b1;
      if (tx_start) begin
        tx      <= 1'b0; // Start bit (0)
        sh_reg  <= tx_data;
        bit_cnt <= 4'd1;
      end
    end else if (bit_cnt >= 4'd1 && bit_cnt <= 4'd8) begin
      tx      <= sh_reg[0]; // Data bits LSB first
      sh_reg  <= {1'b0, sh_reg[7:1]};
      bit_cnt <= bit_cnt + 1'b1;
    end else if (bit_cnt == 4'd9) begin
      tx      <= 1'b1; // Stop bit (1)
      bit_cnt <= 4'd0; // Complete
    end
  end
endmodule

Architectural Deep Dive & Implementation Walkthrough
UART is an asynchronous serial protocol. The receiver detects the falling edge of the Start bit (0), samples data at the midpoint of each baud period, and validates the frame with the Stop bit (1). In real systems, a baud rate generator divides the system clock down to standard baud rates (115200 bps).

Silicon & Hardware Engineering Notes
🛠️ ASIC vs FPGA Hardware Considerations
This educational design assumes 1 cycle per baud bit. Real UARTs include a 16x oversampling baud divider to align the receiver to the center of incoming bit pulses.
