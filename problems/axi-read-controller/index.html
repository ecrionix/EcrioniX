module top_module (
  input  wire        ACLK,
  input  wire        ARESETn,
  input  wire        start_read,
  input  wire [31:0] rd_addr,
  output reg         read_done,
  output reg  [31:0] read_data_out,
  output reg         read_data_valid,
  output reg         read_err,
  output reg         ARVALID,
  input  wire        ARREADY,
  output reg  [31:0] ARADDR,
  input  wire        RVALID,
  output reg         RREADY,
  input  wire [31:0] RDATA,
  input  wire        RLAST,
  input  wire [1:0]  RRESP
);

  reg ar_done;

  always @(posedge ACLK or negedge ARESETn) begin
    if (!ARESETn) begin
      read_done       <= 1'b0;
      read_data_out   <= 32'h0;
      read_data_valid <= 1'b0;
      read_err        <= 1'b0;
      ARVALID         <= 1'b0;
      ARADDR          <= 32'h0;
      RREADY          <= 1'b0;
      ar_done         <= 1'b0;
    end else begin
      read_done       <= 1'b0;
      read_data_valid <= 1'b0;

      // Start Read transaction
      if (start_read && !ARVALID && !ar_done) begin
        ARVALID <= 1'b1;
        ARADDR  <= rd_addr;
        ar_done <= 1'b0;
      end

      // AR Channel handshake
      if (ARVALID && ARREADY) begin
        ARVALID <= 1'b0;
        ar_done <= 1'b1;
        RREADY  <= 1'b1; // Ready to receive read beats
      end

      // R Channel beats
      if (RREADY && RVALID) begin
        read_data_out   <= RDATA;
        read_data_valid <= 1'b1;
        if (RRESP != 2'b00) begin
          read_err <= 1'b1;
        end
        if (RLAST) begin
          RREADY    <= 1'b0;
          ar_done   <= 1'b0;
          read_done <= 1'b1;
        end
      end
    end
  end

endmodule

Architectural Deep Dive & Implementation Walkthrough
AXI4 Read Channel Operation


In AXI4, read transactions use two separate channels:


    
  
  • Address Read (AR) Channel: The master sends the starting address (ARADDR) and burst attributes (ARLEN, ARSIZE, ARBURST).
    • 
  
  • Read Data (R) Channel: The slave returns streaming beats along with error status (RRESP). The final beat of the burst is flagged with RLAST = 1.
    • 



Unlike write transactions, there is no separate response channel for reads; response status (RRESP) is bundled alongside each individual data beat on the R channel.


      
Common RTL Designer Pitfalls
⚠️ Common Mistakes to Avoid
Deasserting RREADY before RLAST arrives, causing slave deadlocks or dropped burst beats.


      
Silicon & Hardware Engineering Notes
🛠️ ASIC vs FPGA Hardware Considerations
AXI read controllers in DDR memory subsystems issue multiple read addresses concurrently (pipelined AR) to saturate memory bank parallelism.
