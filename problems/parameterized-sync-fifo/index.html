module top_module #(
  parameter DATA_WIDTH = 32,
  parameter DEPTH      = 8
)(
  input  wire                  clk,
  input  wire                  rst_n,
  input  wire                  wr_en,
  input  wire [DATA_WIDTH-1:0] wr_data,
  input  wire                  rd_en,
  output reg  [DATA_WIDTH-1:0] rd_data,
  output wire                  full,
  output wire                  empty,
  output reg  [3:0]            count
);

  reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
  reg [2:0] wr_ptr, rd_ptr;

  assign full  = (count == DEPTH);
  assign empty = (count == 4'd0);

  wire valid_wr = wr_en && (!full || rd_en);
  wire valid_rd = rd_en && !empty;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      wr_ptr  <= 3'd0;
      rd_ptr  <= 3'd0;
      count   <= 4'd0;
      rd_data <= {DATA_WIDTH{1'b0}};
    end else begin
      if (valid_wr) begin
        mem[wr_ptr] <= wr_data;
        wr_ptr      <= (wr_ptr + 3'd1) % DEPTH;
      end

      if (valid_rd) begin
        rd_data <= mem[rd_ptr];
        rd_ptr  <= (rd_ptr + 3'd1) % DEPTH;
      end

      case ({valid_wr, valid_rd})
        2'b10: count <= count + 4'd1;
        2'b01: count <= count - 4'd1;
        default: ; // 2'b00 or 2'b11 -> hold count
      endcase
    end
  end

endmodule

Architectural Deep Dive & Implementation Walkthrough
Single-Clock FIFO Architecture


Synchronous FIFOs decouple data rate mismatches between modules running in the same clock domain.


Corner-Case Handling


    
  
  • Simultaneous Read/Write when Full: A well-designed FIFO allows a write when full if a read is concurrently popping an entry. The write overwrites the slot being freed, maintaining count = DEPTH without dropping data.
    • 
  
  • Underflow Protection: If rd_en is asserted when empty = 1, read pointer advancement must be suppressed.
    • 



      
Common RTL Designer Pitfalls
⚠️ Common Mistakes to Avoid
Allowing count to overflow beyond DEPTH or underflow below 0 on invalid enable strobes.


      
Silicon & Hardware Engineering Notes
🛠️ ASIC vs FPGA Hardware Considerations
For small depths (DEPTH <= 32), synthesize to distributed flip-flops; for larger depths (DEPTH >= 64), infer block RAMs (BRAMs).
