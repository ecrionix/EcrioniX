module top_module(
  input        clk,
  input        areset_n,
  input  [3:0] req,
  output reg [3:0] grant
);
  reg [1:0] last_grant;

  function [3:0] get_grant;
    input [1:0] last;
    input [3:0] r;
    reg [3:0] g;
    begin
      g = 4'b0000;
      case (last)
        2'd0: begin
          if (r[1]) g = 4'b0010;
          else if (r[2]) g = 4'b0100;
          else if (r[3]) g = 4'b1000;
          else if (r[0]) g = 4'b0001;
        end
        2'd1: begin
          if (r[2]) g = 4'b0100;
          else if (r[3]) g = 4'b1000;
          else if (r[0]) g = 4'b0001;
          else if (r[1]) g = 4'b0010;
        end
        2'd2: begin
          if (r[3]) g = 4'b1000;
          else if (r[0]) g = 4'b0001;
          else if (r[1]) g = 4'b0010;
          else if (r[2]) g = 4'b0100;
        end
        default: begin // 2'd3
          if (r[0]) g = 4'b0001;
          else if (r[1]) g = 4'b0010;
          else if (r[2]) g = 4'b0100;
          else if (r[3]) g = 4'b1000;
        end
      endcase
      get_grant = g;
    end
  endfunction

  always @(posedge clk or negedge areset_n) begin
    if (!areset_n) begin
      grant      <= 4'b0000;
      last_grant <= 2'd3;
    end else begin
      grant <= get_grant(last_grant, req);
      if (get_grant(last_grant, req) != 4'b0000) begin
        case (get_grant(last_grant, req))
          4'b0001: last_grant <= 2'd0;
          4'b0010: last_grant <= 2'd1;
          4'b0100: last_grant <= 2'd2;
          4'b1000: last_grant <= 2'd3;
        endcase
      end
    end
  end
endmodule

Timing Constraints & Synthesizable RTL Rules

      

        
⚡ Synthesizable Implementation Guidelines

        

          
Timing Contract: Grant updates on `posedge clk`. Exactly one active grant output `grant[3:0]` is asserted when at least one request is active.

          
Synthesis Constraints: Priority rotates circularly to the client immediately following the client that was just granted.

        

      


      
Architectural Deep Dive & Implementation Walkthrough
In a fixed-priority arbiter, high-priority clients can completely starve low-priority clients. A round-robin arbiter ensures 100% fairness by advancing the highest-priority token to $(k+1) \pmod 4$ whenever client $k$ is serviced. This guarantees an upper bound on waiting latency for all bus masters.


      
Common RTL Designer Pitfalls
⚠️ Common Mistakes to Avoid
['Granting multiple clients simultaneously when requests overlap.', 'Advancing priority when no client is requesting, breaking fair cycle order.']


      
Silicon & Hardware Engineering Notes
🛠️ ASIC vs FPGA Hardware Considerations
In wide arbiters (8 to 64 ports), round-robin logic is synthesized using dual priority-encoder chains ($req \& \sim mask$ and $req$) to eliminate $O(N)$ sequential case branches.
